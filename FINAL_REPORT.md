```dart
// Возвращает правильную иконку
IconData _getActionIcon() {
  if (_isSending) return Icons.autorenew;
  if (_textController.text.trim().isNotEmpty || _attachedFilePath != null) return Icons.send;
  if (_speechService?.isAvailable ?? false) {
    return _isListening ? Icons.stop : Icons.mic;
  }
  return Icons.mic_off;
}

// Возвращает правильный цвет
Color _getActionColor(ThemeData theme) {
  if (_isSending) return Colors.white;
  if (_textController.text.trim().isNotEmpty || _attachedFilePath != null) return Colors.white;
  if (_speechService?.isAvailable ?? false) {
    return _isListening ? Colors.red : theme.iconTheme.color ?? Colors.black;
  }
  return Colors.red;
}

// Обрабатывает клик по микрофону
Future<void> _handleMicrophoneAction() async {
  if (_speechService?.isAvailable ?? false) {
    await _startSpeechToText();
    return;
  }
  
  final available = await _speechService?.checkAvailability() ?? false;
  
  if (available) {
    await _startSpeechToText();
  } else {
    SnackbarUtils.showErrorSnackBar(
      context: context,
      message: 'Микрофон недоступен',
      icon: Icons.mic_off,
    );
  }
}
```

**Обновлено:**
- Mobile layout button: `_startSpeechToText` → `_handleMicrophoneAction`
- Desktop layout button: `_startSpeechToText` → `_handleMicrophoneAction`
- Использует `_getActionIcon()` и `_getActionColor()` вместо сложной логики

## 🚀 Как это работает

### Сценарий 1: Пользователь открывает чат
1. Поле ввода пустое
2. Проверяется доступность микрофона
3. Если доступен → 🎤
4. Если НЕ доступен → ⚠️

### Сценарий 2: Пользователь нажимает на 🎤
1. Запускается распознавание речи
2. Текст появляется в поле
3. Иконка меняется на ⏹️

### Сценарий 3: Пользователь нажимает на ⚠️
1. Проверяется доступность микрофона
2. Если разрешено → 🎤
3. Если отказано → ❌ Ошибка через Snackbar

### Сценарий 4: Пользователь вводит текст вручную
1. Иконка автоматически меняется на ✈️
2. По клику → Отправка сообщения

### Сценарий 5: Пользователь прикрепляет файл
1. Иконка меняется на ✈️
2. По клику → Отправка с файлом
